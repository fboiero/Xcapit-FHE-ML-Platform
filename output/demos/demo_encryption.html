import numpy as np
from xcapit_fhe_ml import create_context, encrypt_vector

# Datos originales del usuario
datos = np.array([3.14159, 2.71828, 1.41421, 0.57721])
print(f"Datos originales: {datos}")
2

Crear Contexto Criptográfico CKKS

# Configurar parámetros CKKS
context = create_context(
    preset="ml_optimized",
    poly_modulus_degree=8192,
    scale=2**40
)

print("Contexto CKKS creado exitosamente")
8192
Grado Polinomio (N)
240
Factor de Escala (Δ)
128-bit
Nivel de Seguridad
3

Encriptar Datos

# Encriptar el vector de datos
datos_encriptados = encrypt_vector(context, datos)

print(f"Tipo: {type(datos_encriptados)}")
print(f"Tamaño del ciphertext: {datos_encriptados.size()}")
Ciphertext (lo que ve el servidor):
ct = (c₀, c₁) = ([0x7a8f3c2d1e5b9a4f, 0x2c8e1f7d4a6b3c9e, 0x5f1a2b3c4d5e6f7a, ...], [0x9d4e7f2a1b3c5d8e, 0x6a3f8c2e1d7b4a9f, 0x3c7d1e5f2a6b8c4d, ...])

# Solo el cliente puede descifrar
datos_descifrados = decrypt_vector(context, datos_encriptados)
