Garantia de Privacidad: Los datos encriptados con CKKS son semanticamente seguros. Sin la clave secreta, es computacionalmente imposible obtener informacion sobre los datos originales.

Uso del SDK

from sdk.encryption import CKKSEncryptor, SecurityLevel from sdk.models import LogisticRegression, FHELinearRegression, FHEDecisionTree from sdk.utils.data_loader import SecureDataLoader # 1. Crear encriptador encryptor = CKKSEncryptor(security_level=SecurityLevel.BITS_128) # 2. Cargar y encriptar datos loader = SecureDataLoader(encryptor) encrypted_data = loader.encrypt_dataset(X_train, y_train) # 3. Entrenar modelo sobre datos cifrados model = LogisticRegression(encryptor=encryptor) model.fit(encrypted_data, epochs=100) # 4. Predecir sobre nuevos datos cifrados encrypted_test = loader.encrypt_matrix(X_test) predictions_enc = model.predict(encrypted_test) # 5. Desencriptar solo el resultado predictions = encryptor.decrypt(predictions_enc)
