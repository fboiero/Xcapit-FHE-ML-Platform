Aceptar Invitación
Auto Activate

Endpoint: POST /api/v2/consortiums/invitations/{id}/accept/

Automatización: Signal detecta cuando members ≥ min_members y activa el consorcio automáticamente.

// Signal: auto_activate_consortium
@receiver(post_save, sender=ConsortiumMember)
def auto_activate_consortium(sender, instance, **kwargs):
    if instance.status == 'active':
        consortium = instance.consortium
        if consortium.status == 'draft':
            active_count = consortium.members.filter(status='active').count()
            if active_count >= consortium.min_members:
                consortium.status = 'active'  // AUTOMÁTICO!
                consortium.save()
